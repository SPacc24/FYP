# ==========================================
# 1. FIRST-TIME PROJECT SETUP
# ==========================================

chmod +x install.sh start.sh

./install.sh

# ==========================================
# 2. ACTIVATE PYTHON ENVIRONMENT
# ==========================================

cd project
python -m venv .venv
source .venv/bin/activate
python -m pip install --upgrade pip setuptools wheel
python -m pip install -r requirements.txt

# ==========================================
# 3. CHECK / CREATE .ENV
# ==========================================

python runtime_env.py

# View current generated secrets

python runtime_env.py --show-secrets

# ==========================================
# 4. MYSQL
# ==========================================

sudo systemctl start mysql
sudo systemctl enable mysql

mysql -u root -p

# Create database/user if not already created:
CREATE DATABASE FYP;
CREATE USER 'autopentest'@'localhost' IDENTIFIED BY '<PASSWORD>';
GRANT ALL PRIVILEGES ON autopentest.* TO 'autopentest'@'localhost';
FLUSH PRIVILEGES;
EXIT;
